CREATE TABLE Korisnik(
KorisnikID INT IDENTITY (1,1) NOT NULL,
Ime NVARCHAR (30) NOT NULL,
Prezime NVARCHAR (40) NOT NULL,
Email NVARCHAR (100) UNIQUE NOT NULL,
Lozinka NVARCHAR (40) NOT NULL,
BrojTelefona NVARCHAR (20) NOT NULL,
DatumRodjenja DATE NOT NULL,
Pol NVARCHAR (10) NOT NULL
	CHECK (Pol IN ('Muški', 'Ženski')),
Uloga NVARCHAR (20) NOT NULL
	CHECK (Uloga IN ('Administrator', 'Korisnik'))
)

GO

(
	KorisnikID ASC 
)
