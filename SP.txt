CREATE PROCEDURE DajKorisnikaPoEmailuILozinci
(
	@Email NVARCHAR(100),
	@Lozinka NVARCHAR(40)

CREATE PROCEDURE DajRealizacijeVezbiZaDanasnjiDatum
(
	
	@DatumRealizacije DATE,
	@KorisnikID INT
	
)
AS
BEGIN
	SELECT 
		R.RealizacijaID,
		R.NazivVezbe,
		R.DatumRealizacije,
		R.BrojSerija,
		R.BrojPonavljanja,
		R.Tezina,
		R.Trajanje,
		T.TipVezbeID,
		T.NazivTipa 
	FROM RealizacijaVezbe R INNER JOIN TipVezbe T
	ON R.TipVezbeID = T.TipVezbeID
	WHERE DatumRealizacije=@DatumRealizacije AND KorisnikID=@KorisnikID
END
GO

CREATE PROCEDURE DajRealizacijuVezbePoID
(
    @RealizacijaID INT,
    @KorisnikID INT
)
AS
BEGIN
    SELECT
        R.RealizacijaID,
        R.NazivVezbe,
        R.DatumRealizacije,
        R.BrojSerija,
        R.BrojPonavljanja,
        R.Tezina,
        R.Trajanje,
        T.TipVezbeID,
        T.NazivTipa
    FROM RealizacijaVezbe R
    INNER JOIN TipVezbe T
        ON R.TipVezbeID = T.TipVezbeID
    WHERE R.RealizacijaID = @RealizacijaID
      AND R.KorisnikID = @KorisnikID
END
GO


CREATE PROCEDURE DajSveRealizacijeVezbi
(
	@KorisnikID INT
)
AS
BEGIN
	SELECT 
		R.RealizacijaID,
		R.NazivVezbe,
		R.DatumRealizacije,
		T.NazivTipa,
		R.BrojSerija,
		R.BrojPonavljanja,
		R.Tezina,
		R.Trajanje 
	FROM 
	RealizacijaVezbe R
	INNER JOIN TipVezbe T
	ON R.TipVezbeID = T.TipVezbeID
	WHERE R.KorisnikID=@KorisnikID
	ORDER BY R.DatumRealizacije DESC
END
GO

CREATE PROCEDURE DajBrojRealizacijaZaMesec
(
	@KorisnikID INT,
	@Mesec INT,
	@Godina INT
)
AS
BEGIN
	SELECT COUNT (*) AS BrojRealizacija 
	FROM RealizacijaVezbe
	WHERE
	KorisnikID=@KorisnikID
	AND MONTH(DatumRealizacije)=@Mesec
	AND YEAR(DatumRealizacije)=@Godina
END
GO

CREATE PROCEDURE DajRealizacijeZaPeriod
(
@KorisnikID INT,
@DatumOd DATE,
@DatumDo DATE
)
AS
BEGIN

SELECT
    R.RealizacijaID,
    R.NazivVezbe,
    R.DatumRealizacije,
    T.NazivTipa,
    R.BrojSerija,
    R.BrojPonavljanja,
    R.Tezina,
    R.Trajanje
FROM RealizacijaVezbe R
INNER JOIN TipVezbe T
ON R.TipVezbeID = T.TipVezbeID
WHERE R.KorisnikID = @KorisnikID
AND R.DatumRealizacije BETWEEN @DatumOd AND @DatumDo
ORDER BY R.DatumRealizacije

END
GO

CREATE PROCEDURE DajClanarinuKorisnika
(
	@KorisnikID iNT
)
AS
BEGIN
	SELECT 
		StatusClanarine,
        DatumAktivacije,
        DatumIsteka,
        Cena,
        Popust,
        ZahtevZaProduzenje
	FROM Clanarina
	WHERE KorisnikId=@KorisnikID
END
GO

CREATE PROCEDURE KreirajClanarinuKorisnika
(
    @KorisnikID INT,
	@Cena DECIMAL (10,2),
	@Popust INT
)
AS
BEGIN
    INSERT INTO Clanarina
    (
        StatusClanarine,
        DatumAktivacije,
        DatumIsteka,
        Cena,
        Popust,
        ZahtevZaProduzenje,
        KorisnikID
    )
    VALUES
    (
        'Aktivna',
        GETDATE(),
        DATEADD(DAY,30,GETDATE()),
        @Cena,
        @Popust,
        0,
        @KorisnikID
    )
END
GO

CREATE PROCEDURE PosaljiZahtevZaProduzenje
(
    @KorisnikID INT
)
AS
BEGIN
    UPDATE Clanarina
    SET ZahtevZaProduzenje = 1
    WHERE KorisnikID=@KorisnikID
END
GO

CREATE PROCEDURE PotvrdiUplatuClanarine
(
    @KorisnikID INT,
	@Cena DECIMAL (10,2),
	@Popust INT
)
AS
BEGIN
    UPDATE Clanarina
    SET 
        StatusClanarine='Aktivna',
        DatumAktivacije=GETDATE(),
        DatumIsteka=DATEADD(DAY,30,GETDATE()),
		Cena=@Cena,
		Popust=@Popust,
        ZahtevZaProduzenje=0
    WHERE KorisnikID=@KorisnikID
END
GO

CREATE PROCEDURE OdbijZahtevZaProduzenje
(
    @KorisnikID INT
)
AS
BEGIN
    UPDATE Clanarina
    SET ZahtevZaProduzenje=0
    WHERE KorisnikID=@KorisnikID
END
GO
